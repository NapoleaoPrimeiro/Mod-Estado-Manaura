RP_puppets_integrateterritory = {
	RP_integrate_noncorestate = {
		icon = GFX_decision_generic_nationalism

        ai_will_do = {
			base = 100
		}

		available = {
			FROM = {
				is_owned_by = ROOT
				NOT = { is_core_of = ROOT }
			}
			custom_trigger_tooltip = {
				tooltip = RP_integrate_noncorestate_trigger
				check_variable = {
					var = RP_integrate_noncorestate_count@TAG
					value = RP_GEN_integration_atatime_counter@TAG
					compare = less_than
				}
			}
		}

		state_target = yes

		target_trigger = {
			FROM = {
				OR = {
					AND = {
						is_owned_by = ROOT
						NOT = { is_core_of = ROOT }
						has_state_flag = RP_territorialdecision_complete
					}
				}
			}
		}

		fire_only_once = yes
        days_remove = 60
		cost = 25
		on_map_mode = map_and_decisions_view

		remove_effect = {
			add_stability = 0.02
			add_to_variable = {
				var = RP_integrate_noncorestate_count@TAG
				value = -1
			}
			FROM = {
				add_core_of = ROOT
			}
		}
		
		complete_effect = {
			add_to_variable = {
				var = RP_integrate_noncorestate_count@TAG
				value = 1
			}
		}
	}
}
