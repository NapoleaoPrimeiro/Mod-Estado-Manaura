optimization_player_to_ai_to_player = {
	optimization_ai_to_player = {

		icon = GFX_decision_eng_trade_unions_demand

		available = {
			is_ai = no
			OR = {
				has_idea = optimization_game_speed_balance
				has_idea = dam_idea
			}

		}

		visible = {
			
			is_ai = no
			OR = {
				has_idea = optimization_game_speed_balance
				has_idea = dam_idea
			}
		}

		cost = 0
		days_remove = 0
		days_re_enable = 0

		complete_effect = {
			if = {
				limit = { has_idea = optimization_game_speed_balance }
				remove_ideas = optimization_game_speed_balance
				set_rule = { #DLC
					can_not_declare_war = no
				}
				country_lock_all_division_template = no
			}
			if = {
				limit = { has_idea = dam_idea }
				remove_ideas = dam_idea
			}
			add_ideas = optimization_player_was_here
		}

	}
	
	optimization_player_to_ai = {

		icon = GFX_decision_generic_civil_support

		available = {
			is_ai = no
			NOT = {
				OR = {
					has_idea = optimization_game_speed_balance
					has_idea = dam_idea
				}
			}
		}

		visible = {
			
			is_ai = no
			NOT = {
				OR = {
					has_idea = optimization_game_speed_balance
					has_idea = dam_idea
				}
			}
		}

		cost = 0
		days_remove = 0
		days_re_enable = 0

		complete_effect = {
			remove_ideas = optimization_player_was_here
			add_ideas = optimization_game_speed_balance
			if = {
				limit = {
					has_game_rule = {
						rule = international_market
						option = no
					}
				}
				add_ideas = dam_idea
			}
			
			set_rule = { #DLC
				can_not_declare_war = yes
			}
			country_lock_all_division_template = yes				
			
		}

	}


}
