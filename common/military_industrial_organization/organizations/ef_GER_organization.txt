			armor_value = 0.05

			reliability = 0.10 
			maximum_speed = 0.10
		}

		available = {
			FROM = {
				mio:GER_bmw_tank_industry_organisation = {

			}

			base = 2

				factor = 1.5

		production_bonus = {
			production_resource_need_factor = -0.1
			production_efficiency_gain_factor = 0.1
		}

		available = {
			FROM = {
				mio:GER_bmw_tank_industry_organisation = {
					has_mio_size > 2
				}
			}
		}

		position = { x=1 y=1 } 
		relative_position_id = GER_mio_trait_german_engineering

		token = GER_mio_trait_hotrolled_armors

			production_efficiency_cap_factor = 0.05
			production_efficiency_gain_factor = 0.05
		}

		available = {
			FROM = {
				mio:GER_bmw_tank_industry_organisation = {
					has_mio_size > 2
				}
			}
		}

		position = { x=-1 y=1 }  
		relative_position_id = GER_mio_trait_german_engineering
		ai_will_do = {
			base = 2
			modifier = {
				factor = 1.5

			}
		}

		token = GER_mio_trait_hoesch_facilities

			custom_effect_tooltip = generic_trait_will_not_be_available_in_other_organizations

			}

			production_capacity_factor = 0.05
