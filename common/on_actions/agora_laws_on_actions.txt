# Agora Laws — bayrak bakimi
# Bazi kanunlarin available/visible bloklari bu bayraklara bakar.
# Bu dosya olmazsa o kanunlar ya hic acilmaz ya da yanlis zamanda gorunur.
on_actions = {

	on_startup = {
		effect = {
			every_country = {
				limit = {
					OR = {
						has_government = communism
						has_government = fascism
					}
				}
				set_country_flag = communist_or_fascist_country
			}
		}
	}

	on_government_change = {
		effect = {
			if = {
				limit = {
					OR = {
						has_government = communism
						has_government = fascism
					}
				}
				set_country_flag = communist_or_fascist_country
			}
			else_if = {
				limit = { has_country_flag = communist_or_fascist_country }
				clr_country_flag = communist_or_fascist_country
			}
		}
	}

	on_ruling_party_change = {
		effect = {
			if = {
				limit = {
					OR = {
						has_government = communism
						has_government = fascism
					}
				}
				set_country_flag = communist_or_fascist_country
			}
			else_if = {
				limit = { has_country_flag = communist_or_fascist_country }
				clr_country_flag = communist_or_fascist_country
			}
		}
	}

	on_weekly = {
		effect = {
			every_country = {
				if = {
					limit = {
						is_ai = yes
						NOT = { has_country_flag = ai_country }
					}
					set_country_flag = ai_country
				}
				else_if = {
					limit = {
						is_ai = no
						has_country_flag = ai_country
					}
					clr_country_flag = ai_country
				}
			}
		}
	}
