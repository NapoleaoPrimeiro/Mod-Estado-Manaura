# ---------- Formables (generiert von _quelle/../formables/build.py) ----------
# Frankreich ist immer genau EINE gegruendete Nation; ein neues Formable ersetzt das vorherige (ausser dem Ewigen Koenigreich).
# Kerngebiete gibt es NUR im historischen Gebiet des jeweiligen Formables (woechentlich auch auf neu eroberte Staaten dort).
# Hat Frankreich das Kernland noch nicht, bleibt die Gruendung vorgemerkt und erfolgt automatisch, sobald es erobert ist.
# Bei der Gruendung: Anspruch auf alle Staaten des Gebiets, die Frankreich noch nicht besitzt.
FRW_drop_formable = {
	if = { limit = { has_idea = FRW_union_francaise_formed } remove_ideas = FRW_union_francaise_formed }
	if = { limit = { has_idea = FRW_european_federation } remove_ideas = FRW_european_federation }
	if = { limit = { has_idea = FRW_western_union_idea } remove_ideas = FRW_western_union_idea }
	if = { limit = { has_idea = FRW_latin_empire_idea } remove_ideas = FRW_latin_empire_idea }
	if = { limit = { has_idea = FRW_eurafrican_federation_idea } remove_ideas = FRW_eurafrican_federation_idea }
	if = { limit = { has_idea = FRW_francia_idea } remove_ideas = FRW_francia_idea }
	if = { limit = { has_idea = FRW_burgundy_idea } remove_ideas = FRW_burgundy_idea }
	if = { limit = { has_idea = FRW_grand_empire_idea } remove_ideas = FRW_grand_empire_idea }
	if = { limit = { has_idea = FRW_double_monarchy_idea } remove_ideas = FRW_double_monarchy_idea }
	if = { limit = { has_idea = FRW_franco_british_union_spirit } remove_ideas = FRW_franco_british_union_spirit }
	if = { limit = { has_idea = FRW_drapeau_noir } remove_ideas = FRW_drapeau_noir }
	if = { limit = { has_idea = FRW_eternal_kingdom_spirit } remove_ideas = FRW_eternal_kingdom_spirit }
}
FRW_clear_pending_formables = {
	clr_country_flag = FRW_union_francaise_pending
	clr_country_flag = FRW_european_federation_pending
	clr_country_flag = FRW_western_union_pending
	clr_country_flag = FRW_latin_empire_pending

	clr_country_flag = FRW_burgundy_pending
	clr_country_flag = FRW_grand_empire_pending

	clr_country_flag = FRW_corsair_pending
	clr_country_flag = FRW_eternal_pending
}
FRW_region_cores_union_francaise = {
	every_owned_state = {
		limit = { NOT = { is_core_of = PREV } OR = { state = 459 state = 460 state = 513 state = 514 state = 694 state = 310 state = 706 } }
		add_core_of = PREV
	}
}
FRW_region_claims_union_francaise = {
	every_state = {
		limit = { NOT = { is_owned_by = PREV } NOT = { is_claimed_by = PREV } OR = { state = 459 state = 460 state = 513 state = 514 state = 694 state = 310 state = 706 } }
		add_claim_by = PREV
	}
}
FRW_do_form_union_francaise = {
	if = {
		limit = { NOT = { has_country_flag = FRW_eternal_formed } NOT = { has_cosmetic_tag = FRW_UNION } }
		FRW_drop_formable = yes
		FRW_clear_pending_formables = yes
		set_country_flag = FRW_nation_formed
		set_cosmetic_tag = FRW_UNION
		add_ideas = FRW_union_francaise_formed
		FRW_region_cores_union_francaise = yes
		FRW_region_claims_union_francaise = yes
		news_event = { id = frw.62 hours = 6 }
	}
}
FRW_form_union_francaise = {
	if = {
		limit = { FRW_can_form_union_francaise = yes }
		FRW_do_form_union_francaise = yes
	}
	else_if = {
		limit = { NOT = { has_country_flag = FRW_eternal_formed } NOT = { has_cosmetic_tag = FRW_UNION } }
		set_country_flag = FRW_union_francaise_pending
	}
}
