ENG_check_if_support_is_valid_parliament_effect = {
	IF = {
		limit = {
			check_variable = {
				ENG_parliament_commons_support_amount_var > 100
			}
		}
		set_variable = {
    		var = ENG_parliament_commons_support_amount_var
    		value = 100
		}
	}
	IF = {
		limit = {
			check_variable = {
				ENG_parliament_lords_support_amount_var > 100
			}
		}
		set_variable = {
    		var = ENG_parliament_lords_support_amount_var
    		value = 100
		}
	}
}
