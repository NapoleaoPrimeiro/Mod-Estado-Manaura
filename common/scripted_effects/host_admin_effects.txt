admin_rebuild_player_list = {
	clear_array = global.admin_players
	every_country = {
		limit = { is_ai = no }
		add_to_array = { array = global.admin_players value = THIS }
	}
	admin_count_selected = yes
	admin_build_country_page = yes
}

admin_set_partner = {
	every_country = {
		limit = { has_country_flag = admin_is_partner }
		clr_country_flag = admin_is_partner
	}
	set_country_flag = admin_is_partner
	set_variable = { global.admin_partner = THIS }
}

admin_build_country_page = {
	clear_array = global.admin_countries
	if = {
		limit = { check_variable = { global.admin_tab = 3 } }
		set_variable = { global.admin_c = 0 }
		set_variable = { global.admin_lo = global.admin_page }
		multiply_variable = { global.admin_lo = 10 }
		every_country = {
			limit = { exists = yes }
			set_variable = { global.admin_d = global.admin_c }
			subtract_from_variable = { global.admin_d = global.admin_lo }
			if = {
				limit = {
					check_variable = { global.admin_d > -1 }
					check_variable = { global.admin_d < 10 }
				}
				add_to_array = { array = global.admin_countries value = THIS }
			}
			add_to_variable = { global.admin_c = 1 }
		}
		set_variable = { global.admin_lo1 = global.admin_lo }
		add_to_variable = { global.admin_lo1 = 1 }
		set_variable = { global.admin_hi_shown = global.admin_lo }
		add_to_variable = { global.admin_hi_shown = 10 }
		set_variable = { global.admin_rest = global.admin_c }
		subtract_from_variable = { global.admin_rest = global.admin_hi_shown }
		if = {
			limit = { check_variable = { global.admin_rest > 0 } }
			set_variable = { global.admin_has_next = 1 }
		}
		else = {
			set_variable = { global.admin_has_next = 0 }
			set_variable = { global.admin_hi_shown = global.admin_c }
		}
	}
}

admin_page_next = {
