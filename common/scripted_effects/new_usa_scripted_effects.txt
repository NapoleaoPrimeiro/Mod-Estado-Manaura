USA_start_NC_civil_war = {
	hidden_effect = {
		if = {
			limit = { not = { has_global_flag = USA_NC_civil_war_started_flag } }
			set_global_flag = USA_NC_civil_war_started_flag

			if = {
				limit = { communism > 0.65 }
				set_global_flag = USA_NC_civil_war_high_support_flag
			}
			else_if = {
				limit = { communism > 0.45 }
				set_global_flag = USA_NC_civil_war_medium_support_flag
			}
			else = {
				set_global_flag = USA_NC_civil_war_low_support_flag
			}

			USA_NC_remove_conservative_resistance = yes
			country_event = { id = usa_new_nc_civil_war_event.1 } # starts the civil war
		}
	}
}

			else = {

			}

			country_event = { id = usa_new_an_civil_war_event.1 } # starts the civil war
		}
	}
}

USA_RR_update_influence = {
	set_temp_variable = { temp = USA_RR_total_influence }
	subtract_from_temp_variable = { temp = USA_RR_current_influence }
	set_variable = { USA_RR_german_influence = temp }
	set_temp_variable = { percen = USA_RR_current_influence }
	divide_temp_variable = { percen = USA_RR_total_influence }
	set_variable = { USA_RR_current_influence_percentage = percen }
}

USA_RR_increase_ties = {
	if = {
		limit = {
			not = { has_country_flag = USA_RR_close_influence }
		}
		set_country_flag = USA_RR_close_influence
	}
	else_if = {
		limit = {
			has_country_flag = {
				flag = USA_RR_close_influence
				value = 1
			}
		}
		set_country_flag = {
			flag = USA_RR_close_influence
			value = 2
		}				
	}
	else_if = {
		limit = {
			has_country_flag = {
				flag = USA_RR_close_influence
				value = 2
			}
		}
		set_country_flag = {
			flag = USA_RR_close_influence
			value = 3
		}				
	}
}
