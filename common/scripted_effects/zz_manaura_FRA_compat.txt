# Manaura: o FRA_scripted_effects.txt do mod nao tem este efeito, mas os on_actions vanilla (tfv/goe) chamam ele.
FRA_sync_levant_cosmetic_tag_effect = {
	if = {
		limit = {
			OR = {
				has_cosmetic_tag = SYR_mandate
				has_cosmetic_tag = SYR_vichy
				has_cosmetic_tag = LEB_mandate
				has_cosmetic_tag = LEB_vichy
				AND = {
					is_subject = yes
					OVERLORD = {
						original_tag = FRA
					}
					NOT = { has_cosmetic_tag = SYR_EYALET }
				}
			}
		}
		if = {
			limit = {
				is_subject = yes
				OVERLORD = {
					original_tag = FRA
					is_vichy_france = yes
				}
			}
			if = {
				limit = {
					original_tag = SYR
				}
				set_cosmetic_tag = SYR_vichy
			}
			else = {
				set_cosmetic_tag = LEB_vichy
			}
		}
		else_if = {
			limit = {
				is_subject = yes
				OVERLORD = {
					original_tag = FRA
					NOT = { is_vichy_france = yes }
				}
			}
			if = {
				limit = {
					original_tag = SYR
				}
				set_cosmetic_tag = SYR_mandate
			}
			else = {
				set_cosmetic_tag = LEB_mandate
			}
		}
		else = {
			drop_cosmetic_tag = yes
		}
	}
}
