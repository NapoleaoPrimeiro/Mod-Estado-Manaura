scripted_gui = {
	# lista de divisoes do exercito (interface/unitview.gui -> army_division_list_view)

		window_name = "manaura_no_delete_division_list_window"
		parent_window_window = "army_division_list_view"

		visible = {
			has_war = yes
		}

		effects = {

		}

		triggers = {

	}

		}

		effects = {

		}

		}
	}
