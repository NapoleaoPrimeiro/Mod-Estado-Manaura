# Recalculate in the selected state's scope when the tooltip is requested.
# Uses the shared price helper; no persistent state is written by the tooltip.
defined_text = {
    name = GetCBCoringCurrentCost
    text = {
        trigger = {
            cb_calculate_state_prices = yes
            always = yes
        }
        localization_key = CB_CURRENT_COST_VALUE
    }
}
