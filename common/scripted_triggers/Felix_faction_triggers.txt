		has_government = fascism
	}

		AND = {
			is_in_faction_with = GER
			GER = { is_faction_leader = yes }
		}
		AND = {
			OR = {
				tag = ROM
				tag = HUN
				tag = BUL
				tag = ITA
				tag = GER
			}
			has_government = fascism
			NOT = {has_war_with = GER}
		}
    }
}
