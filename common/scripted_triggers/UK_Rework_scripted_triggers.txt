scripted_trigger = {
	name = ENG_HAS_POLITICAL_PARTY_IN_CHARGE
    parameters = {
        @political_party
    }
    trigger = {
        IF = {
            limit = {
                @political_party = conservatives
            }
            OR = {
                has_country_leader = {
                    character = ENG_winston_churchill
                    ruling_only = yes
                }
                has_country_leader = {
                    character = ENG_stanley_baldwin
                    ruling_only = yes
                }
                has_country_leader = {
                    character = ENG_neville_chamberlain
                    ruling_only = yes
                }
                has_country_leader = {
                    name = "Lord Edward Halifax"
                    ruling_only = yes
                }
            }
        }
        IF = {
            limit = {
                @political_party = labor
            }
            has_country_leader = {
                character = ENG_clement_attlee
                ruling_only = yes
            }
        }
        IF = {
            limit = {
                @political_party = liberals
            }
            has_country_leader = {
                name = "Archibald Sinclair"
                ruling_only = yes
            }
        }
    }
}
