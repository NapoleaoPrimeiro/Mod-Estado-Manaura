# Iberia: intervention in the Spanish Civil War (ROOT = France)
# ---------------------------------------------------------
country_event = {
	id = frw.90
	title = frw.90.t
	desc = frw.90.d
	picture = GFX_report_event_french_tanks
	is_triggered_only = yes

	option = {	# back the Republic (non-fascist side), fight the Nationalists
		name = frw.90.a
		trigger = { any_country = { original_tag = SPR has_civil_war = yes has_government = fascism } }
		ai_chance = { base = 50 modifier = { factor = 0 has_government = fascism } }
		set_country_flag = FRW_iberian_intervention
		every_country = {
			limit = { original_tag = SPR has_civil_war = yes NOT = { has_government = fascism } }
			set_country_flag = FRW_french_backed
		}
		every_country = {
			limit = { original_tag = SPR has_civil_war = yes has_government = fascism NOT = { has_war_with = ROOT } }
			ROOT = { declare_war_on = { target = PREV type = annex_everything } }
		}
	}
	option = {	# back the Nationalists, fight the Republic
		name = frw.90.b
		trigger = { any_country = { original_tag = SPR has_civil_war = yes has_government = fascism } }
		ai_chance = { base = 50 modifier = { factor = 0 has_government = communism } modifier = { factor = 0 has_government = democratic } }
		set_country_flag = FRW_iberian_intervention
		every_country = {
			limit = { original_tag = SPR has_civil_war = yes has_government = fascism }
			set_country_flag = FRW_french_backed
		}
		every_country = {
			limit = { original_tag = SPR has_civil_war = yes NOT = { has_government = fascism } NOT = { has_war_with = ROOT } }
			ROOT = { declare_war_on = { target = PREV type = annex_everything } }
		}
	}
	option = {	# fallback: no fascist side in this civil war -> back the original government
		name = frw.90.c
		trigger = { NOT = { any_country = { original_tag = SPR has_civil_war = yes has_government = fascism } } }
		set_country_flag = FRW_iberian_intervention
		SPR = { set_country_flag = FRW_french_backed }
		every_country = {
			limit = { original_tag = SPR has_civil_war = yes NOT = { tag = SPR } NOT = { has_war_with = ROOT } }
			ROOT = { declare_war_on = { target = PREV type = annex_everything } }
		}
	}
