# The fleet at Mers-el-Kebir
country_event = {
	id = frw_sys.10
	title = frw_sys.10.t
	desc = frw_sys.10.d
	picture = GFX_report_event_french_british_officers
	is_triggered_only = yes
	option = {
		name = frw_sys.10.a
		custom_effect_tooltip = frw_sys.10.a.tt
		hidden_effect = {
			set_temp_variable_to_random = { var = roll min = 0 max = 100 }
			set_temp_variable = { c = FRW_ff_prestige }
			multiply_temp_variable = { c = 0.4 }
			add_to_temp_variable = { c = 20 }
			if = {
				limit = { check_variable = { roll < c } }
				every_country = {
					limit = { FRW_is_vichy = yes }
					transfer_units_fraction = { target = ROOT size = 0 stockpile_ratio = 0 army_ratio = 0 air_ratio = 0 navy_ratio = 0.35 }
				}
				add_to_variable = { FRW_ff_prestige = 5 }
				country_event = { id = frw_sys.11 days = 3 }
			}
			else = {
				add_to_variable = { FRW_ff_prestige = -5 }
				country_event = { id = frw_sys.12 days = 3 }
			}
			FRW_ff_update = yes
		}
		ai_chance = { factor = 60 }
	}
	option = {
		name = frw_sys.10.b
		trigger = { ENG = { exists = yes has_war_with = GER } }
		custom_effect_tooltip = frw_sys.10.b.tt
		hidden_effect = {
			every_country = {
				limit = { FRW_is_vichy = yes }
				destroy_ships = { type = battleship count = 1 }
				destroy_ships = { type = battle_cruiser count = 1 }
				destroy_ships = { type = destroyer count = 3 }
			}
		}
		custom_effect_tooltip = FRW_ffp_tt_m10 hidden_effect = { add_to_variable = { FRW_ff_prestige = -10 } FRW_ff_update = yes }
		ai_chance = { factor = 25 }
	}
	option = {
		name = frw_sys.10.c
		custom_effect_tooltip = FRW_ffp_tt_p5 hidden_effect = { add_to_variable = { FRW_ff_prestige = 5 } FRW_ff_update = yes }
		ai_chance = { factor = 15 }
	}
}
