country_event = { #Fate of Bohemia
    id = RP_austrian_unification.1
    title = RP_austrian_unification.1.t
    desc = RP_austrian_unification.1.d
    picture = GFX_report_event_AUS_political_rally

    is_triggered_only = yes

    option = { #Give it to Osterreich
        name = RP_austrian_unification.1.a

        set_country_flag = RP_bohemia_austria_question

        every_state = {
            limit = {
                is_owned_by = KCZ
            }
            transfer_state_to = KAU
            add_core_of = KAU
        }
    }

    option = { #Form it's own crown
        name = RP_austrian_unification.1.b

        set_country_flag = RP_bohemia_austria_question

        if = {
            limit = {
                KCZ = {
                    has_cosmetic_tag = KCZ_czechoslovakia
                }
            }
            KCZ = {
                set_cosmetic_tag = KCZ_czechoslovakia_independent_crown   
            }
        }

        else = {
            KCZ = {
                set_cosmetic_tag = KCZ_independent_crown
            }
        }
    }

    option = { #Keep it independent
        name = RP_austrian_unification.1.c
    }
}
