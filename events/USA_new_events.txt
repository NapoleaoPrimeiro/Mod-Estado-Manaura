 add_namespace = new_usa_dynamism
  country_event = { # starts process of dynamism unlocked
	id = new_usa_dynamism.1

	hidden = yes

	is_triggered_only = yes

	trigger = {
		has_completed_focus = USA_the_west
		has_completed_focus = USA_the_east
		has_idea = free_trade
	}

	immediate = {
		if = {
			limit = {
				NOT = { has_completed_focus = USA_dynamism_unlocked }
			}
			activate_shine_on_focus = USA_dynamism_unlocked
			set_country_flag = { flag = USA_dynamism_in_progress_flag days = @dynamism_unlocked_timer value = 1 }
			country_event = { id = new_usa_dynamism.2 days = @dynamism_unlocked_timer } # finish
		}
	}
  }
  country_event = { # finish dynamism unlocked
	id = new_usa_dynamism.2

	hidden = yes

	is_triggered_only = yes

	trigger = {
		has_completed_focus = USA_the_west
		has_completed_focus = USA_the_east
		has_idea = free_trade
	}

	immediate = {
		if = {
			limit = {
				has_shine_effect_on_focus = USA_dynamism_unlocked
			}
			complete_national_focus = {
				focus = USA_dynamism_unlocked
				use_side_message = yes
				originator_name = "the USA"
			}
			clr_country_flag = USA_dynamism_in_progress_flag
		}
	}
  }
