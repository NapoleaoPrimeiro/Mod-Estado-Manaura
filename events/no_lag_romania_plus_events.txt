# Accepted Romanian terms. ROOT is Romania and FROM is the Soviet Union.
country_event = {
	id = NO_LAG_ROM_peace.4
	title = NO_LAG_ROM_peace.4.t
	desc = NO_LAG_ROM_peace.4.desc
	picture = GFX_report_event_generic_sign_treaty3
	fire_only_once = yes
	is_triggered_only = yes

	option = {
		name = NO_LAG_ROM_peace.4.a
		custom_effect_tooltip = NO_LAG_ROM_transnistrian_peace_terms_tt

		hidden_effect = {
			# Only Romanian faction members that are actually fighting the USSR
			# are removed from that conflict. Other concurrent wars are untouched.
			if = {
				limit = { is_faction_leader = yes }
				every_other_country = {
					limit = {
						is_in_faction_with = ROOT
						has_war_with = FROM
					}
					white_peace = FROM
				}
			}
			white_peace = FROM

			if = {
				limit = { 192 = { is_owned_by = FROM } }
				transfer_state = 192
				192 = { remove_claim_by = FROM }
			}
			if = {
				limit = { 834 = { is_owned_by = FROM } }
				transfer_state = 834
				834 = { remove_claim_by = FROM }
			}
			if = {
				limit = { 198 = { is_owned_by = FROM } }
				transfer_state = 198
				198 = { remove_claim_by = FROM }
			}

			FROM = {
				add_ai_strategy = {
					type = antagonize
					id = ROOT
					value = -200
				}
			}
		}
	}
}
